 import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Solution {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        List<List<Integer>> lines = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int d = scanner.nextInt(); // Number of elements in this line
            List<Integer> line = new ArrayList<>();
            for (int j = 0; j < d; j++) {
                line.add(scanner.nextInt());
            }
            lines.add(line);
        }

        int q = scanner.nextInt(); 
        for (int i = 0; i < q; i++) {
            int x = scanner.nextInt(); // Line index (1-based)
            int y = scanner.nextInt(); // Position index (1-based)
            int lineIndex = x - 1;
            int posIndex = y - 1;
            if (lineIndex >= 0 && lineIndex < lines.size()) {
                List<Integer> targetLine = lines.get(lineIndex);
                if (posIndex >= 0 && posIndex < targetLine.size()) {
                    System.out.println(targetLine.get(posIndex));
                    continue;
                }
            }

            System.out.println("ERROR!");
        }

        scanner.close();
    }
}
