import java.util.*;

public class test {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        Deque<Integer> deque = new ArrayDeque<>();
        Map<Integer, Integer> map = new HashMap<>();
        
        int n = in.nextInt();
        int m = in.nextInt();
        int maxUnique = 0;

        for (int i = 0; i < n; i++) {
            int num = in.nextInt();
            
            
            deque.add(num);
            map.put(num, map.getOrDefault(num, 0) + 1);
            
           
            if (deque.size() == m) {
                maxUnique = Math.max(maxUnique, map.size());
                
                
                if (maxUnique == m) {
                    System.out.println(maxUnique);
                    return;
                }
                
               
                int oldVal = deque.removeFirst();
                if (map.get(oldVal) == 1) {
                    map.remove(oldVal);
                } else {
                    map.put(oldVal, map.get(oldVal) - 1);
                }
            }
        }
        
        System.out.println(maxUnique);
    }
}
